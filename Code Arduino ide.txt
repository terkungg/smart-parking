// =====================================================
// Smart Parking IoT - ESP8266 + Firebase Realtime Database
// ส่งสถานะช่องจอดขึ้น Firebase ให้หน้าเว็บ (index.html / app.js) อ่าน
//
// โครงสร้างข้อมูลที่เขียนไปที่ /parking
//   A1..A4 : { occupied: true/false }
//   totalSlots : 4
//   online : true
//   lastSeen : เวลาจากเซิร์ฟเวอร์ Firebase (มิลลิวินาที, Unix time)
// =====================================================

#include <ESP8266WiFi.h>
#include <WiFiClientSecure.h>
#include <ESP8266HTTPClient.h>
#include <ESP8266WebServer.h>

// =====================================================
// ตั้งค่า (แก้ตรงนี้)
// =====================================================

const char* WIFI_SSID     = "ชื่อ WiFi ของคุณ";
const char* WIFI_PASSWORD = "รหัส WiFi ของคุณ";

// ต้องตรงกับ FIREBASE_URL ใน app.js
const char* FIREBASE_HOST =
  "smart-parking-iot-6467d-default-rtdb.asia-southeast1.firebasedatabase.app";

// ส่ง heartbeat ทุกกี่ ms (ต้องน้อยกว่า ESP_TIMEOUT_MS = 10000 ใน app.js)
const unsigned long HEARTBEAT_MS = 3000;

// กันเซ็นเซอร์สั่น: สถานะต้องคงที่นานเท่านี้ถึงจะถือว่าเปลี่ยน
const unsigned long DEBOUNCE_MS = 300;


// =====================================================
// GPIO
// =====================================================

#define WIFI_LED   5    // D1
#define OBJECT_LED 16   // D0

const uint8_t SLOT_COUNT = 4;
const uint8_t IR_PINS[SLOT_COUNT] = { 4, 14, 12, 13 };  // D2, D5, D6, D7
const char*   SLOT_NAMES[SLOT_COUNT] = { "A1", "A2", "A3", "A4" };


// =====================================================
// ตัวแปร
// =====================================================

ESP8266WebServer server(80);

bool occupied[SLOT_COUNT]      = { false, false, false, false };
bool lastRaw[SLOT_COUNT]       = { false, false, false, false };
unsigned long rawChangedAt[SLOT_COUNT] = { 0, 0, 0, 0 };

bool dirty = true;                    // มีข้อมูลที่ยังไม่ได้ส่ง
unsigned long lastPush = 0;           // ส่งสำเร็จล่าสุด
unsigned long lastAttempt = 0;        // พยายามส่งล่าสุด
unsigned long lastWiFiTry = 0;
bool firebaseOk = false;
int  lastHttpCode = 0;


// =====================================================
// อ่านเซ็นเซอร์ (IR: LOW = มีรถ) พร้อม debounce
// คืนค่า true ถ้าสถานะช่องใดเปลี่ยน
// =====================================================

bool readSensors() {

  bool changed = false;
  bool anyCar = false;

  for (uint8_t i = 0; i < SLOT_COUNT; i++) {

    bool raw = (digitalRead(IR_PINS[i]) == LOW);

    if (raw != lastRaw[i]) {
      lastRaw[i] = raw;
      rawChangedAt[i] = millis();
    }

    if (raw != occupied[i] && millis() - rawChangedAt[i] >= DEBOUNCE_MS) {
      occupied[i] = raw;
      changed = true;

      Serial.print(occupied[i] ? "🚗 " : "🟢 ");
      Serial.print(SLOT_NAMES[i]);
      Serial.println(occupied[i] ? " : มีรถ" : " : ว่าง");
    }

    if (occupied[i]) anyCar = true;
  }

  digitalWrite(OBJECT_LED, anyCar ? HIGH : LOW);

  return changed;
}


// =====================================================
// ส่งข้อมูลขึ้น Firebase (PATCH = อัปเดตเฉพาะ key ที่ส่ง)
// =====================================================

bool pushToFirebase() {

  String body = "{";

  for (uint8_t i = 0; i < SLOT_COUNT; i++) {
    body += "\"";
    body += SLOT_NAMES[i];
    body += "\":{\"occupied\":";
    body += occupied[i] ? "true" : "false";
    body += "},";
  }

  body += "\"totalSlots\":4,";
  body += "\"online\":true,";
  // ให้เซิร์ฟเวอร์ Firebase ใส่เวลาให้ (ไม่ต้องใช้ NTP และตรงกับ Date.now() ในเว็บ)
  body += "\"lastSeen\":{\".sv\":\"timestamp\"}";
  body += "}";

  WiFiClientSecure client;
  client.setInsecure();               // ไม่ตรวจใบรับรอง (ใช้งานง่ายบน ESP8266)
  client.setBufferSizes(1024, 512);   // ประหยัด RAM

  HTTPClient http;
  http.setTimeout(5000);

  String url = String("https://") + FIREBASE_HOST + "/parking.json";

  if (!http.begin(client, url)) {
    Serial.println("❌ http.begin ล้มเหลว");
    return false;
  }

  http.addHeader("Content-Type", "application/json");

  lastHttpCode = http.sendRequest("PATCH", body);

  http.end();

  if (lastHttpCode == 200) {
    return true;
  }

  Serial.print("❌ Firebase ตอบกลับ: ");
  Serial.println(lastHttpCode);   // 401 = Rules ไม่อนุญาตให้เขียน, ค่าติดลบ = เชื่อมต่อไม่ได้
  return false;
}


// =====================================================
// หน้าเว็บในตัวบอร์ด (แบบย่อ ไว้ debug)  http://<IP ของบอร์ด>/
// =====================================================

void handleRoot() {

  String s = "Smart Parking ESP8266\n\n";

  for (uint8_t i = 0; i < SLOT_COUNT; i++) {
    s += SLOT_NAMES[i];
    s += " : ";
    s += occupied[i] ? "occupied" : "free";
    s += "\n";
  }

  s += "\nFirebase : ";
  s += firebaseOk ? "OK" : "FAIL";
  s += " (HTTP ";
  s += lastHttpCode;
  s += ")\nIP : ";
  s += WiFi.localIP().toString();
  s += "\nRSSI : ";
  s += WiFi.RSSI();
  s += " dBm\n";

  server.send(200, "text/plain; charset=UTF-8", s);
}


// =====================================================
// WiFi
// =====================================================

void connectWiFi() {

  Serial.println();
  Serial.println("กำลังเชื่อมต่อ WiFi...");

  WiFi.mode(WIFI_STA);
  WiFi.setAutoReconnect(true);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  unsigned long start = millis();

  while (WiFi.status() != WL_CONNECTED && millis() - start < 20000) {
    digitalWrite(WIFI_LED, !digitalRead(WIFI_LED));
    Serial.print(".");
    delay(500);
    yield();
  }

  if (WiFi.status() == WL_CONNECTED) {
    digitalWrite(WIFI_LED, HIGH);
    Serial.println();
    Serial.print("✅ WiFi เชื่อมต่อสำเร็จ  IP: ");
    Serial.println(WiFi.localIP());
  }
  else {
    digitalWrite(WIFI_LED, LOW);
    Serial.println();
    Serial.println("❌ WiFi เชื่อมต่อไม่สำเร็จ (จะลองใหม่อัตโนมัติ)");
  }
}


// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(115200);
  delay(500);

  Serial.println();
  Serial.println("================================");
  Serial.println("  SMART PARKING -> FIREBASE");
  Serial.println("================================");

  pinMode(WIFI_LED, OUTPUT);
  pinMode(OBJECT_LED, OUTPUT);
  digitalWrite(WIFI_LED, LOW);
  digitalWrite(OBJECT_LED, LOW);

  for (uint8_t i = 0; i < SLOT_COUNT; i++) {
    pinMode(IR_PINS[i], INPUT);
    bool raw = (digitalRead(IR_PINS[i]) == LOW);
    lastRaw[i] = raw;
    occupied[i] = raw;         // อ่านค่าเริ่มต้นก่อนส่งครั้งแรก
  }

  connectWiFi();

  server.on("/", handleRoot);
  server.begin();
}


// =====================================================
// LOOP
// =====================================================

void loop() {

  server.handleClient();

  // อ่านเซ็นเซอร์ ถ้าเปลี่ยนให้ส่งทันที
  if (readSensors()) {
    dirty = true;
  }

  // ดูแล WiFi (ไม่บล็อกการทำงาน)
  if (WiFi.status() != WL_CONNECTED) {

    digitalWrite(WIFI_LED, LOW);

    if (millis() - lastWiFiTry >= 5000) {
      lastWiFiTry = millis();
      Serial.println("⚠️ WiFi หลุด กำลังต่อใหม่...");
      WiFi.reconnect();
    }
  }
  else {

    digitalWrite(WIFI_LED, HIGH);

    bool heartbeatDue = (millis() - lastPush >= HEARTBEAT_MS);
    bool canTry       = (millis() - lastAttempt >= 500);

    if ((dirty || heartbeatDue) && canTry) {

      lastAttempt = millis();
      firebaseOk = pushToFirebase();

      if (firebaseOk) {
        lastPush = millis();
        dirty = false;
      }
    }
  }

  yield();
}
