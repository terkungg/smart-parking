bool slotA4 = false;


// =====================================================
// อ่าน Sensor
// =====================================================

void readSensors() {

  slotA1 = (digitalRead(IR_A1) == LOW);
  slotA2 = (digitalRead(IR_A2) == LOW);
  slotA3 = (digitalRead(IR_A3) == LOW);
  slotA4 = (digitalRead(IR_A4) == LOW);


  // ===================================================
  // LED แสดงสถานะรวม
  // ถ้ามีรถช่องใดช่องหนึ่ง → LED ติด
  // ===================================================

  if (slotA1 || slotA2 || slotA3 || slotA4) {

    digitalWrite(OBJECT_LED, HIGH);

  }
  else {

    digitalWrite(OBJECT_LED, LOW);

  }


  // ===================================================
  // Serial แสดงเฉพาะตอนสถานะเปลี่ยน
  // ===================================================

  static bool oldA1 = false;
  static bool oldA2 = false;
  static bool oldA3 = false;
  static bool oldA4 = false;


  if (slotA1 != oldA1) {

    if (slotA1)
      Serial.println("🚗 A1 : มีรถ");
    else
      Serial.println("🟢 A1 : ว่าง");

    oldA1 = slotA1;
  }


  if (slotA2 != oldA2) {

    if (slotA2)
      Serial.println("🚗 A2 : มีรถ");
    else
      Serial.println("🟢 A2 : ว่าง");

    oldA2 = slotA2;
  }


  if (slotA3 != oldA3) {

    if (slotA3)
      Serial.println("🚗 A3 : มีรถ");
    else
      Serial.println("🟢 A3 : ว่าง");

    oldA3 = slotA3;
  }


  if (slotA4 != oldA4) {

    if (slotA4)
      Serial.println("🚗 A4 : มีรถ");
    else
      Serial.println("🟢 A4 : ว่าง");

    oldA4 = slotA4;
  }
}

  return card;
}

// =====================================================
// เชื่อมต่อ WiFi
// =====================================================

bool connectWiFi() {

  Serial.println();
  Serial.println("กำลังเชื่อมต่อ WiFi...");

  WiFi.mode(WIFI_STA);

  WiFi.begin(ssid, password);


  unsigned long startTime = millis();


  while (
    WiFi.status() != WL_CONNECTED &&
    millis() - startTime < 20000
  ) {

    digitalWrite(WIFI_LED, !digitalRead(WIFI_LED));

    Serial.print(".");

    delay(500);

    yield();
  }


  if (WiFi.status() == WL_CONNECTED) {

    digitalWrite(WIFI_LED, HIGH);

    Serial.println();
    Serial.println("================================");
    Serial.println("✅ WiFi เชื่อมต่อสำเร็จ");
    Serial.println("================================");

    Serial.print("IP Address: ");
    Serial.println(WiFi.localIP());

    Serial.print("Signal: ");
    Serial.print(WiFi.RSSI());
    Serial.println(" dBm");

    return true;
  }


  digitalWrite(WIFI_LED, LOW);

  Serial.println();
  Serial.println("❌ WiFi เชื่อมต่อไม่สำเร็จ");

  return false;
}


// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(115200);

  delay(500);


  Serial.println();
  Serial.println("================================");
  Serial.println("     IoT PARKING SYSTEM");
  Serial.println("================================");


  // ===================================================
  // Pin Mode
  // ===================================================

  pinMode(WIFI_LED, OUTPUT);

  pinMode(IR_A1, INPUT);

  pinMode(IR_A2, INPUT);

  pinMode(IR_A3, INPUT);

  pinMode(IR_A4, INPUT);

  pinMode(OBJECT_LED, OUTPUT);


  digitalWrite(WIFI_LED, LOW);

  digitalWrite(OBJECT_LED, LOW);


  // ===================================================
  // WiFi
  // ===================================================

  connectWiFi();


  // ===================================================
  // Web Server
  // ===================================================

  server.on("/", handleRoot);

  server.begin();


  Serial.println();
  Serial.println("================================");
  Serial.println("Web Server STARTED");

  if (WiFi.status() == WL_CONNECTED) {

    Serial.print("เปิดเว็บ: http://");
    Serial.println(WiFi.localIP());

  }

  Serial.println("================================");
}


// =====================================================
// LOOP
// =====================================================

void loop() {

  // Web Server
  server.handleClient();


  // อ่าน IR
  readSensors();


  // ===================================================
  // ตรวจ WiFi ทุก 5 วินาที
  // ===================================================

  static unsigned long lastWiFiCheck = 0;


  if (millis() - lastWiFiCheck >= 5000) {

    lastWiFiCheck = millis();


    if (WiFi.status() != WL_CONNECTED) {

      Serial.println();
      Serial.println("⚠️ WiFi หลุด");

      connectWiFi();

    }
    else {

      digitalWrite(WIFI_LED, HIGH);

    }
  }


  yield();
}
